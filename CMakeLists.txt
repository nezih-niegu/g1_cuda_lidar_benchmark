cmake_minimum_required(VERSION 3.16)
project(g1_cuda_lidar_benchmark LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2 REQUIRED)
option(BUILD_CUDA "Build the CUDA filtering backends" ON)
if(BUILD_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  set(CMAKE_CUDA_STANDARD 17)
  set(CMAKE_CUDA_STANDARD_REQUIRED ON)
  add_library(lidar_cuda src/cuda_filter.cu)
  target_include_directories(lidar_cuda PUBLIC include)
  target_link_libraries(lidar_cuda PUBLIC CUDA::cudart)
  set_target_properties(lidar_cuda PROPERTIES CUDA_SEPARABLE_COMPILATION ON POSITION_INDEPENDENT_CODE ON)
  target_compile_options(lidar_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
endif()
add_executable(lidar_mapper src/mapper.cpp)
target_include_directories(lidar_mapper PRIVATE include)
ament_target_dependencies(lidar_mapper rclcpp sensor_msgs std_msgs tf2_ros tf2)
if(BUILD_CUDA)
  target_compile_definitions(lidar_mapper PRIVATE ENABLE_CUDA=1)
  target_link_libraries(lidar_mapper lidar_cuda)
  target_include_directories(lidar_mapper PRIVATE ${CUDAToolkit_INCLUDE_DIRS})
endif()
install(TARGETS lidar_mapper DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/detect_ros_distro.sh scripts/setup_workspace.sh scripts/sensor_relay.py scripts/synthetic_lidar.py scripts/summarize.py scripts/summary_across_runs.py scripts/run_three_modes.sh scripts/collect_system_info.sh scripts/profile_mapper.sh DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config docs DESTINATION share/${PROJECT_NAME})
ament_package()
